Excel & Google Sheets Lead Manager

Customer Inquiries & Excel Export

Download customer messages as an Excel (.csv) file or connect a live Google Sheet so every inquiry saves automatically.

1. Connect Live Google Sheet (Auto-Save Every Customer to Excel)

Follow these 3 quick steps once so every customer from any phone or computer goes straight into your spreadsheet:

  1. Open sheets.new to create a new Google Sheet, then click Extensions → Apps Script.
  2. Delete any code there, paste the script on the right, and click Deploy → New deployment.
  3. Choose type Web app, set Who has access to Anyone, click Deploy, and paste the Web app URL below (or set googleSheetWebhookUrl in src/lib/site-data.ts).

2. Copy Google Apps Script Code

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Date & Time",
      "Full Name",
      "Phone / WhatsApp",
      "Email",
      "Website Type",
      "Budget Range",
      "Country",
      "Project Details"
    ]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date().toLocaleString("en-IN", { timeZone: "Asia/Kolkata" }),
    data.name || "",
    data.phone || "",
    data.email || "",
    data.businessType || "",
    data.budgetRange || "",
    data.selectedCountry || "",
    data.projectDetails || ""
  ]);
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
